import axios from 'axios';
import * as cheerio from 'cheerio';

export default async function handler(req, res) {
  // 1. Configuración de cabeceras CORS
  res.setHeader('Access-Control-Allow-Credentials', 'true');
  res.setHeader('Access-Control-Allow-Origin', '*'); // Permite peticiones desde cualquier origen (Android / Frontend)
  res.setHeader('Access-Control-Allow-Methods', 'GET, OPTIONS');
  res.setHeader(
    'Access-Control-Allow-Headers',
    'X-CSRF-Token, X-Requested-With, Accept, Accept-Version, Content-Length, Content-MD5, Content-Type, Date, X-Api-Version'
  );

  // Manejo de petición Preflight OPTIONS para CORS
  if (req.method === 'OPTIONS') {
    return res.status(200).end();
  }

  // Validar que solo se permitan peticiones GET
  if (req.method !== 'GET') {
    return res.status(405).json({ error: 'Método no permitido. Utiliza GET.' });
  }

  // 2. Extraer y validar el parámetro 'causa'
  const { causa } = req.query;

  if (!causa) {
    return res.status(400).json({
      error: 'Falta el parámetro obligatorio "causa". Ejemplo: /api/satje?causa=09281-2023-01452'
    });
  }

  try {
    // URL del portal de la Función Judicial (e-SATJE / CJA)
    const url = `https://procesosjudiciales.funcionjudicial.gob.ec/busqueda-causa?numero_causa=${encodeURIComponent(causa)}`;

    // 3. Realizar la petición HTTP simulando un navegador real
    const response = await axios.get(url, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
        'Accept-Language': 'es-ES,es;q=0.9',
      },
      timeout: 10000 // Timeout de 10 segundos
    });

    // 4. Analizar el HTML devuelto usando cheerio
    const $ = cheerio.load(response.data);

    // Búsqueda de las filas de la tabla de actuaciones procesales
    // Ajusta los selectores según la tabla dinámica o estructura renderizada por e-SATJE
    const filasTabla = $('table.tabla-actuaciones tbody tr, table tr').filter((_, el) => {
      return $(el).find('td').length >= 2; // Filtrar filas que contengan celdas reales
    });

    if (filasTabla.length === 0) {
      return res.status(404).json({
        error: 'No se encontraron actuaciones procesales o la causa no existe.'
      });
    }

    // 5. Extraer la última actuación (última fila de la tabla)
    const ultimaFila = filasTabla.last();
    const celdas = ultimaFila.find('td');

    // Función auxiliar para limpiar espacios, saltos de línea y tabulaciones
    const cleanText = (element) => element.text().replace(/\s+/g, ' ').trim();

    // Mapeo flexible de celdas
    const fechaHoraRaw = cleanText($(celdas[0]));
    const tipoActuacion = cleanText($(celdas[1]));
    const extractoProvidencia = celdas.length > 2 ? cleanText($(celdas[2])) : tipoActuacion;

    // Formatear texto final
    const textoActuacion = tipoActuacion !== extractoProvidencia && extractoProvidencia 
      ? `${tipoActuacion}: ${extractoProvidencia}`
      : (extractoProvidencia || tipoActuacion);

    // 6. Retornar la respuesta JSON esperada
    return res.status(200).json({
      numero_causa: causa,
      ultima_actuacion_fecha: fechaHoraRaw || 'Fecha no disponible',
      ultima_actuacion_texto: textoActuacion || 'Sin extracto disponible'
    });

  } catch (error) {
    console.error('Error al consultar e-SATJE:', error.message);

    return res.status(500).json({
      error: 'Error interno al consultar el portal e-SATJE.',
      detalle: error.message
    });
  }
}
